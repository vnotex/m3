# Match the real core's sources/linkage, but keep replacement allocators and
# their control symbols out of the installed library and public package.
get_target_property(_m3_fault_sources m3core SOURCES)
get_target_property(_m3_source_dir m3core SOURCE_DIR)
list(TRANSFORM _m3_fault_sources PREPEND "${_m3_source_dir}/")
if(M3_BUILD_SHARED)
    add_library(m3_fault_core SHARED ${_m3_fault_sources} failing_allocator.cpp)
    target_compile_definitions(m3_fault_core PRIVATE M3_BUILD_DLL PUBLIC M3_SHARED)
else()
    add_library(m3_fault_core STATIC ${_m3_fault_sources} failing_allocator.cpp)
endif()
set_target_properties(m3_fault_core PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO
    POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(m3_fault_core PUBLIC ../include)
target_include_directories(m3_fault_core SYSTEM PRIVATE ../third_party)
m3_target_defaults(m3_fault_core)
add_executable(m3_test_failures test_failures.cpp)
target_link_libraries(m3_test_failures PRIVATE m3_fault_core)
target_include_directories(m3_test_failures SYSTEM PRIVATE ../third_party)
set_target_properties(m3_test_failures PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
m3_target_defaults(m3_test_failures)
if(MINGW)
    # A dynamic libstdc++ keeps some std::string allocations inside its own DLL,
    # bypassing replacement new. Bind those calls to the test allocator as well.
    target_link_options(m3_fault_core PRIVATE -static-libstdc++ -static-libgcc)
    target_link_options(m3_test_failures PRIVATE -static-libstdc++ -static-libgcc)
endif()
foreach(case outputs mutations)
    add_test(NAME m3_allocations_${case} COMMAND m3_test_failures ${case})
    set_tests_properties(m3_allocations_${case} PROPERTIES TIMEOUT 120)
endforeach()
